## uEnv.txt for booting the Yocto image from microSD (FAT32, partition 1).
## Read by the Bitmain U-Boot 2013.04: bootcmd -> mmc rescan -> loadbootenv/importbootenv ->
## uenvcmd. Only this file and the three images below are needed on the card; NAND is untouched.
## Files next to this one (copy them from repo/mainline/out/):
##   uImage-yocto.bin, am335x-antminer-breakout.dtb (or another profile), antminer-image.cpio.gz.u-boot
##
## Memory map (same as tools/netboot.ps1, proven): uImage at 0x82000000, DTB 0x88000000,
## initramfs 0x88100000; bootm relocates DTB and initramfs itself.

bootfile=uImage-yocto.bin
fdtfile=am335x-antminer-breakout.dtb
initramfsfile=antminer-image.cpio.gz.u-boot
mmcpart=1

loadaddr=0x82000000
fdtaddr=0x88000000
initramfsaddr=0x88100000

## mainline: OMAP UARTs are 8250 -> ttyS0 (the old env has ttyO0)
console=ttyS0,115200n8
optargs=

sdargs=setenv bootargs console=${console} ${optargs} init=/sbin/init
loadfdt=load mmc ${mmcdev}:${mmcpart} ${fdtaddr} ${fdtfile}
loadinitramfs=load mmc ${mmcdev}:${mmcpart} ${initramfsaddr} ${initramfsfile}
loaduimage=load mmc ${mmcdev}:${mmcpart} ${loadaddr} ${bootfile}
bootsd=bootm ${loadaddr} ${initramfsaddr} ${fdtaddr}
uenvcmd=echo Booting Yocto image from SD; run loadfdt loadinitramfs loaduimage sdargs bootsd
